
"use client";

import Link from "next/link";
import { useState } from "react";
import {
  CheckCircle2,
  ChevronDown,
  Copy,
  ExternalLink,
  LogOut,
  RefreshCw,
  Wallet,
  X,
} from "lucide-react";

import { useWeb3Wallet } from "@/hooks/useWeb3Wallet";
import { explorerAddressUrl } from "@/lib/web3/config";

function shortAddress(address: string) {
  return `${address.slice(0, 6)}...${address.slice(-4)}`;
}

export default function Topbar() {
  const wallet = useWeb3Wallet();

  const [menuOpen, setMenuOpen] = useState(false);
  const [error, setError] = useState("");
  const [copied, setCopied] = useState(false);

  const connected = Boolean(
    wallet.address && wallet.provider
  );

  const onArc = connected && wallet.isOnArcTestnet;

  async function connectWallet() {
    setError("");

    try {
      await wallet.connect();
      setMenuOpen(false);
    } catch (caught) {
      setError(
        caught instanceof Error
          ? caught.message
          : "Could not connect wallet."
      );
    }
  }

  async function connectWithWallet(
    option: (typeof wallet.wallets)[number]
  ) {
    setError("");

    try {
      await wallet.connectWith(option);
      setMenuOpen(false);
    } catch (caught) {
      setError(
        caught instanceof Error
          ? caught.message
          : "Could not connect wallet."
      );
    }
  }

  async function copyAddress() {
    if (!wallet.address) return;

    try {
      await navigator.clipboard.writeText(
        wallet.address
      );
      setCopied(true);
    } catch {
      setError("Could not copy wallet address.");
    }
  }

  function disconnectWallet() {
    wallet.disconnect();
    setMenuOpen(false);
    setError("");
    setCopied(false);
  }

  return (
    <header className="relative z-30 flex min-h-20 items-center justify-between gap-3 border-b border-slate-200 bg-white px-4 py-3 sm:px-6 lg:px-8">
      {/* Logo */}
      <Link
        href="/dashboard"
        className="min-w-0"
      >
        <h1 className="text-2xl font-bold tracking-tight text-blue-600">
          FlowUSD
        </h1>

        <p className="text-xs text-slate-500">
          USDC Payments on Arc
        </p>
      </Link>

      {/* Wallet Controls */}
      <div className="flex items-center gap-3">
        <span className="hidden rounded-full bg-blue-50 px-3 py-2 text-xs font-semibold text-blue-700 sm:inline-flex">
          Arc Testnet
        </span>

        <div className="relative">
          <button
            type="button"
            onClick={() => {
              setMenuOpen((previous) => !previous);
              setError("");
              setCopied(false);
            }}
            className={`inline-flex items-center gap-2 rounded-xl border px-3 py-3 text-sm font-semibold transition sm:px-4 ${
              onArc
                ? "border-emerald-200 bg-emerald-50 text-emerald-700 hover:bg-emerald-100"
                : connected
                  ? "border-amber-200 bg-amber-50 text-amber-700 hover:bg-amber-100"
                  : "border-blue-600 bg-blue-600 text-white hover:bg-blue-700"
            }`}
            aria-expanded={menuOpen}
            aria-label="Wallet menu"
          >
            {onArc ? (
              <CheckCircle2 size={17} />
            ) : (
              <Wallet size={17} />
            )}

            <span>
              {wallet.address
                ? shortAddress(wallet.address)
                : "Connect Wallet"}
            </span>

            <ChevronDown size={15} />
          </button>

          {menuOpen && (
            <>
              {/* Background to close the dropdown */}
              <button
                type="button"
                aria-label="Close wallet menu"
                onClick={() => setMenuOpen(false)}
                className="fixed inset-0 z-40 cursor-default bg-transparent"
              />

              <div className="absolute right-0 top-full z-50 mt-3 w-[min(350px,calc(100vw-32px))] rounded-2xl border border-slate-200 bg-white p-5 shadow-xl">
                <div className="mb-4 flex items-center justify-between">
                  <h2 className="font-bold text-slate-900">
                    {connected
                      ? "Connected Wallet"
                      : "Connect Wallet"}
                  </h2>

                  <button
                    type="button"
                    onClick={() => setMenuOpen(false)}
                    className="rounded-lg p-1 text-slate-500 hover:bg-slate-100"
                    aria-label="Close"
                  >
                    <X size={18} />
                  </button>
                </div>

                {connected && wallet.address ? (
                  <div className="space-y-4">
                    <div className="rounded-xl bg-slate-50 p-3">
                      <p className="mb-2 text-xs text-slate-500">
                        Wallet Address
                      </p>

                      <p className="break-all font-mono text-xs text-slate-800">
                        {wallet.address}
                      </p>
                    </div>

                    <div className="flex flex-wrap items-center justify-between gap-2">
                      <span
                        className={`rounded-full px-3 py-1 text-xs font-semibold ${
                          onArc
                            ? "bg-green-100 text-green-700"
                            : "bg-amber-100 text-amber-700"
                        }`}
                      >
                        {onArc
                          ? "Arc Testnet Connected"
                          : "Wrong Network"}
                      </span>

                      <button
                        type="button"
                        onClick={copyAddress}
                        className="inline-flex items-center gap-2 text-xs font-semibold text-blue-600 hover:underline"
                      >
                        <Copy size={13} />
                        {copied
                          ? "Copied"
                          : "Copy Address"}
                      </button>
                    </div>

                    {onArc && (
                      <div className="rounded-xl bg-blue-50 p-4">
                        <p className="text-xs text-slate-500">
                          On-chain USDC Balance
                        </p>

                        <div className="mt-2 flex items-center justify-between gap-2">
                          <p className="break-all text-lg font-bold text-slate-900">
                            {wallet.loadingBalance
                              ? "Loading..."
                              : wallet.balance !== null
                                ? `${wallet.balance} USDC`
                                : "Unavailable"}
                          </p>

                          <button
                            type="button"
                            disabled={wallet.loadingBalance}
                            onClick={() =>
                              void wallet.refreshBalance()
                            }
                            className="rounded-lg p-2 text-blue-600 hover:bg-blue-100 disabled:opacity-50"
                            title="Refresh USDC balance"
                          >
                            <RefreshCw
                              size={16}
                              className={
                                wallet.loadingBalance
                                  ? "animate-spin"
                                  : ""
                              }
                            />
                          </button>
                        </div>
                      </div>
                    )}

                    {!onArc && (
                      <p className="rounded-xl bg-amber-50 p-3 text-xs text-amber-800">
                        Please switch your wallet to
                        Arc Testnet (Chain ID 5042002).
                      </p>
                    )}

                    <a
                      href={explorerAddressUrl(
                        wallet.address
                      )}
                      target="_blank"
                      rel="noopener noreferrer"
                      className="flex items-center justify-center gap-2 rounded-xl border border-slate-300 px-4 py-3 text-sm font-semibold text-slate-700 hover:bg-slate-50"
                    >
                      <ExternalLink size={16} />
                      View on Arc Explorer
                    </a>

                    <button
                      type="button"
                      onClick={disconnectWallet}
                      className="flex w-full items-center justify-center gap-2 rounded-xl border border-red-200 px-4 py-3 text-sm font-semibold text-red-600 hover:bg-red-50"
                    >
                      <LogOut size={16} />
                      Disconnect Wallet
                    </button>
                  </div>
                ) : (
                  <div className="space-y-3">
                    <p className="text-sm text-slate-500">
                      Select a browser wallet to use
                      FlowUSD on Arc Testnet.
                    </p>

                    {!wallet.discoveryDone ? (
                      <p className="text-sm text-slate-500">
                        Detecting wallets...
                      </p>
                    ) : wallet.wallets.length === 0 ? (
                      <p className="rounded-xl bg-amber-50 p-3 text-sm text-amber-800">
                        No compatible browser wallet
                        was found.
                      </p>
                    ) : (
                      <div className="space-y-2">
                        {wallet.wallets.map((option) => (
                          <button
                            type="button"
                            key={option.uuid}
                            disabled={wallet.connecting}
                            onClick={() =>
                              void connectWithWallet(
                                option
                              )
                            }
                            className="flex w-full items-center gap-3 rounded-xl border border-slate-200 px-4 py-3 text-left text-sm font-semibold text-slate-800 hover:border-blue-300 hover:bg-blue-50 disabled:opacity-50"
                          >
                            <Wallet size={17} />
                            {option.name}
                          </button>
                        ))}
                      </div>
                    )}

                    {wallet.wallets.length === 1 && (
                      <button
                        type="button"
                        disabled={wallet.connecting}
                        onClick={() =>
                          void connectWallet()
                        }
                        className="w-full rounded-xl bg-blue-600 px-4 py-3 text-sm font-semibold text-white hover:bg-blue-700 disabled:opacity-50"
                      >
                        {wallet.connecting
                          ? "Connecting..."
                          : "Connect"}
                      </button>
                    )}
                  </div>
                )}

                {error && (
                  <p
                    role="alert"
                    className="mt-4 break-words rounded-xl bg-red-50 p-3 text-xs text-red-700"
                  >
                    {error}
                  </p>
                )}
              </div>
            </>
          )}
        </div>
      </div>
    </header>
  );
}
