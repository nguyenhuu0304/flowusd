
"use client";

import Link from "next/link";
import { useState } from "react";
import {
  ArrowDownLeft,
  ArrowUpRight,
  CheckCircle2,
  Copy,
  ExternalLink,
  History,
  Link2,
  RefreshCw,
  Wallet2,
} from "lucide-react";

import { useWeb3Wallet } from "@/hooks/useWeb3Wallet";

import {
  ARC_TESTNET_CHAIN_ID_HEX,
} from "@/lib/web3/config";

function shortAddress(address: string) {
  return `${address.slice(0, 8)}...${address.slice(-6)}`;
}

export default function DashboardContent() {
  const wallet = useWeb3Wallet();

  const [copied, setCopied] = useState(false);
  const [refreshing, setRefreshing] = useState(false);
  const [walletError, setWalletError] = useState("");

  const connected =
    Boolean(wallet.address && wallet.provider);

  const ready =
    connected && wallet.isOnArcTestnet;

  async function copyAddress() {
    if (!wallet.address) return;

    try {
      await navigator.clipboard.writeText(
        wallet.address
      );

      setCopied(true);

      window.setTimeout(() => {
        setCopied(false);
      }, 2000);
    } catch {
      setCopied(false);
    }
  }

  async function connectWallet() {
    setWalletError("");

    try {
      await wallet.connect();
    } catch (error) {
      setWalletError(
        error instanceof Error
          ? error.message
          : "Could not connect wallet."
      );
    }
  }

  async function refreshBalance() {
    if (!ready || refreshing) return;

    setRefreshing(true);

    try {
      await wallet.refreshBalance();
    } finally {
      setRefreshing(false);
    }
  }

  const balanceText =
    ready && wallet.balance !== null
      ? `${wallet.balance} USDC`
      : "— USDC";

  return (
    <main className="flex-1 overflow-y-auto bg-slate-50 px-4 py-8 sm:px-6 lg:px-8">
      <div className="mx-auto max-w-5xl space-y-6">

        {/* Header */}
        <div className="flex flex-wrap items-start justify-between gap-4">
          <div>
            <h1 className="text-3xl font-bold text-slate-900">
              Dashboard
            </h1>

            <p className="mt-2 text-sm text-slate-500">
              Your on-chain payments on Arc Testnet.
            </p>
          </div>

          <span className="inline-flex items-center gap-2 rounded-full bg-blue-50 px-4 py-2 text-sm font-semibold text-blue-700">
            <CheckCircle2 size={16} />
            Arc Testnet
          </span>
        </div>

        {/* Wallet balance */}
        <section className="rounded-2xl border border-slate-200 bg-white p-6 shadow-sm sm:p-8">
          <div className="flex flex-wrap items-start justify-between gap-4">
            <div>
              <div className="flex items-center gap-2 text-sm font-medium text-slate-500">
                <Wallet2 size={18} />
                Connected Wallet
              </div>

              {wallet.address ? (
                <button
                  type="button"
                  onClick={copyAddress}
                  className="mt-3 inline-flex items-center gap-2 break-all text-left font-mono text-sm text-slate-800 hover:text-blue-600"
                  title={wallet.address}
                >
                  {shortAddress(wallet.address)}
                  <Copy size={14} />

                  {copied && (
                    <span className="font-sans text-xs text-green-600">
                      Copied
                    </span>
                  )}
                </button>
              ) : (
                <p className="mt-3 text-sm text-slate-500">
                  No wallet connected
                </p>
              )}
            </div>

            <span
              className={`rounded-full px-3 py-1 text-xs font-semibold ${
                ready
                  ? "bg-green-100 text-green-700"
                  : connected
                  ? "bg-amber-100 text-amber-700"
                  : "bg-slate-100 text-slate-600"
              }`}
            >
              {ready
                ? "Connected"
                : connected
                ? "Wrong network"
                : "Disconnected"}
            </span>
          </div>

          <div className="mt-8">
            <p className="text-sm text-slate-500">
              USDC Balance
            </p>

            <div className="mt-2 flex flex-wrap items-center gap-4">
              <h2 className="break-all text-3xl font-bold tracking-tight text-slate-900 sm:text-4xl">
                {wallet.loadingBalance
                  ? "Loading..."
                  : balanceText}
              </h2>

              {ready && (
                <button
                  type="button"
                  onClick={() => void refreshBalance()}
                  disabled={
                    refreshing || wallet.loadingBalance
                  }
                  className="rounded-lg border border-slate-200 p-2 text-slate-600 hover:bg-slate-50 disabled:opacity-50"
                  title="Refresh USDC balance"
                >
                  <RefreshCw
                    size={18}
                    className={
                      refreshing ? "animate-spin" : ""
                    }
                  />
                </button>
              )}
            </div>

            <p className="mt-3 text-xs text-slate-500">
              Balance is read from the connected
              wallet on Arc Testnet.
            </p>
          </div>

          {!connected && (
            <div className="mt-6">
              {wallet.needsWalletSelection ? (
                <div className="space-y-3">
                  <p className="text-sm text-slate-600">
                    Choose a wallet to connect:
                  </p>

                  <div className="flex flex-wrap gap-2">
                    {wallet.wallets.map((option) => (
                      <button
                        type="button"
                        key={option.uuid}
                        onClick={() => {
                          setWalletError("");

                          wallet
                            .connectWith(option)
                            .catch((error) => {
                              setWalletError(
                                error instanceof Error
                                  ? error.message
                                  : "Wallet connection failed."
                              );
                            });
                        }}
                        disabled={wallet.connecting}
                        className="rounded-xl border border-slate-300 px-4 py-3 text-sm font-semibold hover:bg-slate-50 disabled:opacity-50"
                      >
                        {option.name}
                      </button>
                    ))}
                  </div>
                </div>
              ) : (
                <button
                  type="button"
                  onClick={() => void connectWallet()}
                  disabled={wallet.connecting}
                  className="rounded-xl bg-blue-600 px-5 py-3 text-sm font-semibold text-white hover:bg-blue-700 disabled:opacity-50"
                >
                  {wallet.connecting
                    ? "Connecting..."
                    : "Connect Wallet"}
                </button>
              )}
            </div>
          )}

          {connected && !wallet.isOnArcTestnet && (
            <div className="mt-6 rounded-xl bg-amber-50 p-4 text-sm text-amber-800">
              Your wallet is not connected to
              Arc Testnet. Please switch to
              Chain ID{" "}
              {Number.parseInt(
                ARC_TESTNET_CHAIN_ID_HEX,
                16
              )}.
            </div>
          )}

          {walletError && (
            <p
              role="alert"
              className="mt-4 rounded-xl bg-red-50 p-3 text-sm text-red-700"
            >
              {walletError}
            </p>
          )}

          {/* Send and Receive */}
          <div className="mt-8 grid grid-cols-1 gap-3 border-t border-slate-100 pt-6 sm:grid-cols-2">
            <Link
              href="/send"
              className="flex items-center justify-center gap-3 rounded-xl bg-blue-600 px-5 py-4 font-semibold text-white transition hover:bg-blue-700"
            >
              <ArrowUpRight size={20} />
              Send USDC
            </Link>

            <Link
              href="/receive"
              className="flex items-center justify-center gap-3 rounded-xl border border-slate-300 bg-white px-5 py-4 font-semibold text-slate-900 transition hover:bg-slate-50"
            >
              <ArrowDownLeft size={20} />
              Receive USDC
            </Link>
          </div>
        </section>

        {/* Main features */}
        <div className="grid grid-cols-1 gap-4 sm:grid-cols-2">
          <Link
            href="/payment-links"
            className="group rounded-2xl border border-slate-200 bg-white p-6 shadow-sm transition hover:border-blue-300 hover:shadow-md"
          >
            <div className="flex items-start justify-between">
              <div className="flex h-12 w-12 items-center justify-center rounded-xl bg-blue-50 text-blue-600">
                <Link2 size={23} />
              </div>

              <ExternalLink
                size={18}
                className="text-slate-400 transition group-hover:text-blue-600"
              />
            </div>

            <h3 className="mt-5 text-lg font-bold text-slate-900">
              Payment Links
            </h3>

            <p className="mt-2 text-sm text-slate-500">
              Create, share and manage on-chain
              USDC payment requests.
            </p>
          </Link>

          <Link
            href="/transactions"
            className="group rounded-2xl border border-slate-200 bg-white p-6 shadow-sm transition hover:border-blue-300 hover:shadow-md"
          >
            <div className="flex items-start justify-between">
              <div className="flex h-12 w-12 items-center justify-center rounded-xl bg-emerald-50 text-emerald-600">
                <History size={23} />
              </div>

              <ExternalLink
                size={18}
                className="text-slate-400 transition group-hover:text-blue-600"
              />
            </div>

            <h3 className="mt-5 text-lg font-bold text-slate-900">
              Transactions
            </h3>

            <p className="mt-2 text-sm text-slate-500">
              View confirmed USDC transfers
              from Arc Testnet.
            </p>
          </Link>
        </div>

        {/* Footer */}
        <p className="pb-4 text-center text-xs text-slate-400">
          FlowUSD · Arc Testnet · Non-custodial
          on-chain payments
        </p>
      </div>
    </main>
  );
}
