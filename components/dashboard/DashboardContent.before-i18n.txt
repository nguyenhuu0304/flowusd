
"use client";

import { useState } from "react";
import {
  CheckCircle2,
  History,
  Link2,
  RefreshCw,
  Wallet,
} from "lucide-react";

import { useWeb3Wallet } from "@/hooks/useWeb3Wallet";
import PaymentLinksCard from "@/components/dashboard/PaymentLinksCard";
import TransactionHistory from "@/components/dashboard/TransactionHistory";

type DashboardTab = "payments" | "transactions";

export default function DashboardContent() {
  const wallet = useWeb3Wallet();

  const [activeTab, setActiveTab] =
    useState<DashboardTab>("payments");

  const connected = Boolean(
    wallet.address && wallet.provider
  );

  const ready = connected && wallet.isOnArcTestnet;

  const balance =
    ready && wallet.balance !== null
      ? `${wallet.balance} USDC`
      : "— USDC";

  return (
    <main className="flex-1 bg-slate-50 px-4 py-8 sm:px-6 lg:px-8">
      <div className="mx-auto max-w-5xl space-y-6">
        <div>
          <h1 className="text-3xl font-bold text-slate-900">
            Dashboard
          </h1>

          <p className="mt-2 text-sm text-slate-500">
            Manage your USDC payments on Arc Testnet.
          </p>
        </div>

        <section className="rounded-2xl border border-slate-200 bg-white p-6 shadow-sm">
          <div className="flex flex-wrap items-center justify-between gap-3">
            <div className="flex items-center gap-2 text-sm text-slate-500">
              <Wallet size={18} />
              Connected Wallet
            </div>

            <span
              className={`inline-flex items-center gap-2 rounded-full px-3 py-1 text-xs font-semibold ${
                ready
                  ? "bg-emerald-100 text-emerald-700"
                  : connected
                    ? "bg-amber-100 text-amber-700"
                    : "bg-slate-100 text-slate-600"
              }`}
            >
              <CheckCircle2 size={14} />
              {ready
                ? "Arc Testnet Connected"
                : connected
                  ? "Wrong Network"
                  : "Not Connected"}
            </span>
          </div>

          <div className="mt-5">
            <p className="text-sm text-slate-500">
              On-chain USDC Balance
            </p>

            <div className="mt-2 flex flex-wrap items-center gap-3">
              <h2 className="text-3xl font-bold text-slate-900">
                {wallet.loadingBalance
                  ? "Loading..."
                  : balance}
              </h2>

              {ready && (
                <button
                  type="button"
                  disabled={wallet.loadingBalance}
                  onClick={() =>
                    void wallet.refreshBalance()
                  }
                  title="Refresh Balance"
                  className="rounded-lg border border-slate-200 p-2 text-slate-600 hover:bg-slate-50 disabled:opacity-50"
                >
                  <RefreshCw
                    size={17}
                    className={
                      wallet.loadingBalance
                        ? "animate-spin"
                        : ""
                    }
                  />
                </button>
              )}
            </div>

            {wallet.address && (
              <p className="mt-3 break-all font-mono text-xs text-slate-500">
                {wallet.address}
              </p>
            )}

            {!ready && (
              <p className="mt-3 text-sm text-amber-700">
                Connect a wallet on Arc Testnet
                using the button in the top-right corner.
              </p>
            )}
          </div>
        </section>

        <div
          className="grid grid-cols-2 gap-3"
          role="tablist"
          aria-label="Dashboard sections"
        >
          <button
            type="button"
            role="tab"
            aria-selected={activeTab === "payments"}
            onClick={() => setActiveTab("payments")}
            className={`flex items-center justify-center gap-2 rounded-xl border px-4 py-4 font-semibold transition ${
              activeTab === "payments"
                ? "border-blue-600 bg-blue-600 text-white"
                : "border-slate-200 bg-white text-slate-700 hover:bg-slate-50"
            }`}
          >
            <Link2 size={18} />
            Payment Links
          </button>

          <button
            type="button"
            role="tab"
            aria-selected={activeTab === "transactions"}
            onClick={() => setActiveTab("transactions")}
            className={`flex items-center justify-center gap-2 rounded-xl border px-4 py-4 font-semibold transition ${
              activeTab === "transactions"
                ? "border-blue-600 bg-blue-600 text-white"
                : "border-slate-200 bg-white text-slate-700 hover:bg-slate-50"
            }`}
          >
            <History size={18} />
            Transactions
          </button>
        </div>

        {activeTab === "payments" ? (
          <section role="tabpanel">
            <PaymentLinksCard />
          </section>
        ) : (
          <section role="tabpanel">
            <TransactionHistory />
          </section>
        )}

        <p className="pb-6 text-center text-xs text-slate-400">
          FlowUSD · Arc Testnet · On-chain USDC Payments
        </p>
      </div>
    </main>
  );
}
